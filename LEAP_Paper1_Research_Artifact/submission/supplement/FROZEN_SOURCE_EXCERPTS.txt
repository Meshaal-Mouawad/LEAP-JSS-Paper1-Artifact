209: def _expand_forward_statement(
210:     lines: List[str], start_index: int, back: int = 5, max_forward: int = 200
211: ) -> str:
212:     """Return a forward-expanded SQL-ish statement context starting a few lines before start_index.
213: 
214:     - Includes up to `back` lines before the start_index for context.
215:     - Scans forward up to `max_forward` lines or until a likely statement terminator is seen:
216:       semicolon at low parenthesis depth, a lone GO, or end of file.
217:     - Works for SQL/T-SQL/PLSQL/HANA snippets where KPI comment precedes a CREATE/SELECT.
218:     """
219:     import re as _re
220: 
221:     n = len(lines)
222:     if n == 0:
223:         return ""
224: 
225:     start = max(0, min(start_index, n - 1))
226:     ctx_start = max(0, start - back)
227: 
228:     end = start
229:     depth = 0
230:     for j in range(start, min(n, start + max_forward)):
231:         line = lines[j].rstrip("\n")
232:         end = j
233:         # Track parentheses depth to avoid stopping too early inside CAST(...), CASE(...), etc.
234:         depth += line.count("(") - line.count(")")
235:         if _re.search(r";\s*$", line) and depth <= 0:
236:             break
237:         if _re.match(r"^\s*GO\s*$", line, _re.I):
238:             break
239:         if _re.match(
240:             r"^\s*/\s*$", line
241:         ):  # Oracle '/' statement separator on its own line
242:             break
243: 
244:     return "\n".join(line.rstrip("\n") for line in lines[ctx_start : end + 1])
